// Pines de entrada y señalización
const int ledRojo = 8;
const int ledVerde = 9;
const int boton = 2;

void setup() {
  pinMode(ledRojo, OUTPUT);
  pinMode(ledVerde, OUTPUT);
  pinMode(boton, INPUT); // Pull-down externo
  digitalWrite(ledRojo, HIGH);
  digitalWrite(ledVerde, LOW);
}

void loop() {
  if (digitalRead(boton) == HIGH) {
    digitalWrite(ledRojo, LOW);
    for (int i = 0; i < 3; i++) {
      digitalWrite(ledVerde, HIGH); delay(500);
      digitalWrite(ledVerde, LOW);  delay(500);
    }
    digitalWrite(ledVerde, HIGH); delay(3000);
    digitalWrite(ledVerde, LOW);
    digitalWrite(ledRojo, HIGH);
    while (digitalRead(boton) == HIGH) delay(10);
  }
}
